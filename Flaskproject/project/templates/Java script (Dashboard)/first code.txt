        // Function to update bin statistics
        function updateBinStatistics() {
            // Simulated data for demonstration
            var totalBins = 20;
            var halfFilled = 10;
            var emptyBins = 5;
            var completelyFilled = 5;
    
            // Update the text in bin cards
            document.getElementById('totalBins').innerText = 'Total Bins: ' + totalBins;
            document.getElementById('halfFilled').innerText = 'Half Filled: ' + halfFilled;

            document.getElementById('completelyFilled').innerText = 'Completely Filled: ' + completelyFilled;
            document.getElementById('completelyFilledBins').innerText = 'Completely Filled: ' + completelyFilled;
        }
    
        // Function to show content section
        function showContent(contentId) {
            // Hide all content sections
            var sections = document.getElementsByClassName('content-section');
            for (var i = 0; i < sections.length; i++) {
                sections[i].classList.remove('show');
            }
            // Show the selected content section
            document.getElementById(contentId).classList.add('show');
    
            // If the selected content section is 'half-full', update bin statistics
            if (contentId === 'half-full' || contentId === 'completely-full') {
                updateBinStatistics();
            }
        }
    
        // Function to toggle the dropdown menu
        function toggleDropdown(dropdownId) {
            var dropdown = document.getElementById(dropdownId);
            dropdown.classList.toggle('show');
        }
    
        // Function to generate random latitude and longitude within a specific range
        function getRandomLocation() {
            var lat = 13.0 + Math.random() * 0.2; // Latitude between 13.0 and 13.2
            var lon = 77.5 + Math.random() * 0.2; // Longitude between 77.5 and 77.7
            return [lat, lon];
        }
        function handleMarkerClick(marker, name, type) {
    // Get the table element
    var table = document.getElementById('locationNames');

    // Create a new row
    var row = table.insertRow();

    // Insert cells for type and name
    var cell1 = row.insertCell(0);
    var cell2 = row.insertCell(1);

    // Set the cell values
    cell1.innerHTML = type;
    cell2.innerHTML = name;
}
    
        // Function to show truck locations and bin locations
        function showLocations() {
    var map = L.map('map').setView([13.1155, 77.6070], 13); // Set the initial center and zoom level

    // Add the base map (you can choose different map providers)
    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
        attribution: '© OpenStreetMap contributors'
    }).addTo(map);

    // Define custom red marker icon for trucks
    var redIcon = L.icon({
        iconUrl: 'https://maps.google.com/mapfiles/ms/icons/red-dot.png',
        iconSize: [32, 32], // size of the icon
        iconAnchor: [16, 32], // point of the icon which will correspond to marker's location
        popupAnchor: [0, -32] // point from which the popup should open relative to the iconAnchor
    });

    // Define custom green marker icon for bins
    var greenIcon = L.icon({
        iconUrl: 'https://maps.google.com/mapfiles/ms/icons/green-dot.png',
        iconSize: [32, 32], // size of the icon
        iconAnchor: [16, 32], // point of the icon which will correspond to marker's location
        popupAnchor: [0, -32] // point from which the popup should open relative to the iconAnchor
    });

    // Define truck locations with names
    var truckLocations = [
        { name: 'Truck 1', location: [13.1155, 77.6070] },
        { name: 'Truck 2', location: [13.1747, 77.5485] },
        { name: 'Truck 3', location: [13.1236, 77.5748] }
    ];

    // Add markers for truck locations
    truckLocations.forEach(function (truck) {
        var marker = L.marker(truck.location, { icon: redIcon })
            .bindPopup(truck.name)
            .addTo(map);
        
        // Handle click event on marker
        marker.on('click', function(e) {
            handleMarkerClick(marker, truck.name, 'Truck');
        });
    });

    // Define bin statuses
    var binStatuses = ['Empty', 'Half-Filled', 'Completely Filled'];

    // Generate 20 random bin locations with random statuses
    for (var i = 0; i < 20; i++) {
        var location = getRandomLocation();
        var status = binStatuses[Math.floor(Math.random() * binStatuses.length)];
        var marker = L.marker(location, { icon: greenIcon })
            .bindPopup('Bin ' + (i + 1) + ': ' + status)
            .addTo(map);
        
        // Handle click event on marker
        marker.on('click', function(e) {
            var markerName = e.target.getPopup().getContent().split(':')[0];
            handleMarkerClick(marker, markerName, 'Bin');
        });
    }
}
